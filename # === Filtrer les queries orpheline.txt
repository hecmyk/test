# === Filtrer les queries orphelines (PDF absent après extraction/oversize) ====

pdf_stems = {p.stem for p in pdfs_dir.iterdir() if p.suffix.lower() == ".pdf"}

clean_before = len(clean)
clean = [rec for rec in clean if QID_RE.match(rec["query-id"]).group("doc") in pdf_stems]
n_dropped = clean_before - len(clean)

(DST / "queries_en_clean.json").write_text(
    json.dumps(clean, ensure_ascii=False, indent=2), encoding="utf-8"
)

# Recalculer docs_needed sur les queries filtrées (source de vérité mise à jour)
docs_needed = {QID_RE.match(rec["query-id"]).group("doc") for rec in clean}

log(f"\n[queries] filtered orphan queries (no matching PDF): {n_dropped} dropped")
log(f"    queries_en_clean.json now has {len(clean)} records")
log(f"    referenced docs: {len(docs_needed)}")
