def call_mistral_ocr(image_b64: str, max_retries: int = 2) -> tuple[dict | None, float, str | None]:
    """Call Mistral OCR on a base64-encoded PNG image.

    Retries transient failures (network errors, 429, 5xx) with exponential backoff
    (1s, 2s). 4xx are deterministic and not retried. The backoff sleeps while holding
    the caller's `_MISTRAL_SEM` slot: that back-pressure is intended, so a burst of
    429s throttles the whole pool.

    Args:
        image_b64: Base64-encoded PNG payload.
        max_retries: Number of retries after the initial attempt.

    Returns:
        `(response_json, elapsed_ms, error)`: on success `error` is None; on failure
        `response_json` is None and `error` holds the message. `elapsed_ms` covers the
        last attempt only (backoff sleeps excluded).
    """
    HEADERS = {"Content-Type": "application/json"}
    payload = {
        "model": MISTRAL_OCR_MODEL,
        "document": {
            "type": "image_url",
            "image_url": f"data:image/png;base64,{image_b64}",
        },
    }

    for attempt in range(max_retries + 1):
        start_time = time.time()
        try:
            response = requests.post(MISTRAL_OCR_URL, headers=HEADERS, json=payload, timeout=30)
            elapsed = (time.time() - start_time) * 1000

            if response.status_code == 200:
                return response.json(), elapsed, None

            if response.status_code in (429, 500, 502, 503, 504) and attempt < max_retries:
                time.sleep(2 ** attempt)
                continue

            return None, elapsed, f"API ERROR : {response.status_code} - {response.text}"

        except requests.RequestException as e:
            elapsed = (time.time() - start_time) * 1000
            if attempt < max_retries:
                time.sleep(2 ** attempt)
                continue
            return None, elapsed, str(e)







from functools import lru_cache


@lru_cache(maxsize=1)
def load_predictor() -> pdi.Predictor:
    """Load the PP-DocLayoutV2 model and return its inference predictor.

    Cached: the model is loaded on the first call and reused afterwards, so importing
    this module costs nothing. Not thread-safe (two concurrent first calls would each
    build a predictor); `run_layout` is never called in parallel.

    Returns:
        The initialized predictor.
    """
    config = pdi.Config("PP-DocLayoutV2/inference.pdmodel", "PP-DocLayoutV2/inference.pdiparams")

    config.disable_gpu()

    config.switch_ir_optim(False)
    config.set_cpu_math_library_num_threads(4)

    predictor = pdi.create_predictor(config)

    logger.info("model loaded")
    return predictor

    Args:

        save_images: Also write one annotated image per page next to output_path.
        score_threshold: Minimum score to keep a detection.
        labels: Class-index-to-label list.

    Returns:
        Mapping from page key (`page_N`) to its list of detection dicts.
    """

    output_path = Path(output_path)

    doc = fitz.open(pdf_path)
    doc_json_data = {}

        page = doc[page_idx]

        detections = run_inference(load_predictor(), x, im_shape, scale_factor)
