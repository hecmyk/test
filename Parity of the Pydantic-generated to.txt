"""Parity of the Pydantic-generated tool schemas with the v2.0 literals, dict-equal.

The three tool definitions agent_dom_v2 announces to the endpoint used to be hand-written
JSON literals; since the refactor they are generated from Pydantic models through
protocol._ToolSchema. This test pins the generated schemas to the literals frozen from
the v2.0 file (tool_schemas_v20.json, search_and_read taken with top_k=5): same keys,
same values, same nesting. Key ORDER is not compared — the literals themselves were not
consistent on it, and JSON Schema gives it no meaning.

It also guards against generator drift between Pydantic versions: the runtime's Pydantic
is whatever the openai SDK pulled, so run this wherever the agent runs.

Run:  python -m pytest scratch/refactor/test_tool_schema_parity.py
  or: python scratch/refactor/test_tool_schema_parity.py
"""

import json
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
sys.path.insert(0, str(HERE.parents[1]))
sys.path.insert(0, str(HERE))

FIXTURE = json.loads((HERE / "tool_schemas_v20.json").read_text(encoding="utf-8"))


def _tools():
    try:
        import openai  # noqa: F401  — real SDK available: import the package as is
    except ImportError:
        import harness_v2
        harness_v2._install_stubs()
    from agent_dom_v2.orchestrator import build_delegate_read_tool, build_search_and_read_tool
    from agent_dom_v2.report import build_report_tool
    return {"report": build_report_tool(), "delegate_read": build_delegate_read_tool(),
            "search_and_read": build_search_and_read_tool(5)}


def test_tool_schemas_match_v20_literals():
    generated = _tools()
    for name, expected in FIXTURE.items():
        assert generated[name] == expected, name


def test_no_title_ref_or_default_anywhere():
    blob = json.dumps(_tools())
    for token in ('"title"', '"$ref"', '"$defs"', '"default"', '"anyOf"'):
        assert token not in blob, token


if __name__ == "__main__":
    test_tool_schemas_match_v20_literals()
    test_no_title_ref_or_default_anywhere()
    print("tool schema parity: OK")

FIELD_DEFINITIONS = {"qa": "Answer the question grounded in the document."}

_SEMANTIC_NOTE = (
    "agent v2 (orchestrator + ephemeral subagent). n_steps = LLM calls of BOTH tiers. "
    "get_section_calls = sections PRE-LOADED for the subagent, so used_section means "

def _init_run_v2(run_id: str, notes: str, log_subagent_details: bool,
                 p: float | None, seed: int) -> Path:
    """The run directory, created with its metadata on first use.

    run_metadata.json is written ONCE per run_id: a later call on the same run is a
    continuation, and rewriting it would move created_at away from the code that actually
    produced the earliest rows.

    Args:
        run_id: Name of this benchmark iteration.
        notes: Free text describing what this run changes.
        log_subagent_details: Whether this run dumps the heavy artefacts.
        p: Stratified sampling fraction, or None for the full question list.
        seed: Seed of that draw; recorded so the sample can be rebuilt.

    Returns:
        The eval/mmlongbench/runs/{run_id} directory.
    """
    run_dir = RUNS / run_id
    run_dir.mkdir(parents=True, exist_ok=True)

    metadata = run_dir / "run_metadata.json"
    if not metadata.exists():
        metadata.write_text(json.dumps({
            "run_id": run_id,
            "benchmark": "MMLongBench-Doc",
            "agent": "agent_dom_v2",
            "created_at": datetime.now().isoformat(timespec="seconds"),
            "notes": f"{_SEMANTIC_NOTE} {notes}".strip(),
            "ORCHESTRATOR_MODEL": agent_dom_v2.ORCHESTRATOR_MODEL,
            "SUBAGENT_MODEL": agent_dom_v2.SUBAGENT_MODEL,
            "ORCHESTRATOR_MAX_ROUNDS": agent_dom_v2.ORCHESTRATOR_MAX_ROUNDS,
            "SEARCH_TOP_K": agent_dom_v2.SEARCH_TOP_K,
            "SUBAGENT_MAX_TOOL_CALLS": agent_dom_v2.SUBAGENT_MAX_TOOL_CALLS,
            "LOG_SUBAGENT_DETAILS": log_subagent_details,
            "FIELD_DEFINITIONS": FIELD_DEFINITIONS,
            "prompts": PROMPTS_PATH.name,
            "sample_p": p,
            "sample_seed": seed if p is not None else None,
        }, ensure_ascii=False, indent=2), encoding="utf-8")
    return run_dir


def run_one(dom: dict, skeleton: str, sample: dict, pdf_path: Path | None = None,
            log_subagent_details: bool = False,
            verifications: list | None = None) -> tuple[dict, list[dict]]:
    """Answer one question and build its result row; an agent failure becomes `error`.

    Row shape is run_mm.run_one's, field for field — gold-free, joined by query_id in
    score_mm — so extraction and scoring do not branch on which agent produced it.

    `pred_source_pages` is written EXACTLY as the model sent it, with no coercion: the
    schema is advisory on this endpoint, so a scalar or a malformed entry is a fact about
    the run and metrics.as_pages normalizes it at read time.

    Args:
        dom: The document's DOM.
        skeleton: Its XML skeleton.
        sample: One entry of dataset.load_samples.
        pdf_path: The source PDF, or None to keep the call text-only.
        log_subagent_details: Collect the heavy per-invocation artefacts.
        verifications: Caller-owned sink for the per-verdict records of the agent's
            non-repairing checks, appended in place; None drops them. Only the four
            aggregate keys reach `agent_metrics`.

    Returns:
        `(row, debug)`. debug is empty unless log_subagent_details is on, and never enters
        the row.
    """
    started = time.perf_counter()
    debug: list[dict] = []
    try:
        draft, metrics, debug = run_dom_agent_v2(
            dom, skeleton, FIELD_DEFINITIONS, mode="qa", question=sample["question"],
            pdf_path=pdf_path, prompts=PROMPTS_MM_V2,
            max_rounds=agent_dom_v2.ORCHESTRATOR_MAX_ROUNDS,
            model=agent_dom_v2.ORCHESTRATOR_MODEL,
            subagent_model=agent_dom_v2.SUBAGENT_MODEL,
            search_top_k=agent_dom_v2.SEARCH_TOP_K,
            subagent_max_tool_calls=agent_dom_v2.SUBAGENT_MAX_TOOL_CALLS,
            log_subagent_details=log_subagent_details, verifications=verifications)
        answer, error = draft["qa"], None
    except BadRequestError as e:
        answer, metrics, error = {}, {}, f"{type(e).__name__}: {e}"
    metrics["wall_clock_s"] = round(time.perf_counter() - started, 2)

    return {
        "query_id": sample["query_id"],
        "doc_id": sample["doc_id"],
        "stem": sample["stem"],
        "question": sample["question"],
        "pred_answer": answer.get("answer"),
        "pred_quote": answer.get("quote"),
        "pred_source_pages": answer.get("source_pages"),
        "agent_metrics": metrics,
        "error": error,
    }, debug


def _dump_debug(run_dir: Path, query_id: str, debug: list[dict]) -> None:
    """Write one question's subagent artefacts under the run's debug subdirectory."""
    out_dir = run_dir / DEBUG_DIR_NAME
    out_dir.mkdir(exist_ok=True)
    (out_dir / f"{query_id}.json").write_text(
        json.dumps(debug, ensure_ascii=False, indent=2), encoding="utf-8")


def _dump_verifications(run_dir: Path, query_id: str, verifications: list) -> None:
    """Append one question's verdict records to the run's verifications.jsonl.

    One line per verdict, query_id first: {query_id, check, round, section_id,
    figure | page, verdict} — a tidy table, read back with pd.read_json(lines=True) or, one
    line at a time, agent_dom_v2.verifications.VerdictAdapter.validate_json. Same
    append-and-flush discipline as the per-document results, and written only once the
    row is: an infra failure raises before this point, so a replayed question never leaves
    orphan verdicts behind.
    """
    with (run_dir / VERIFICATIONS_NAME).open("a", encoding="utf-8") as f:
        for record in verifications:
            f.write(record.model_copy(update={"query_id": query_id}).model_dump_json() + "\n")


def run_mm_v2(run_id: str, limit: int | None = None, notes: str = "",
           log_subagent_details: bool = False,
           p: float | None = None, seed: int = 42) -> None:
    """Run every pending question of one v2 iteration, one JSONL file per document.

    A question already present in the run's file is skipped. That is a crash net, not a
    resume mode: to start over, pass a new run_id.

    Args:
        run_id: Name of this iteration; results land in eval/mmlongbench/runs/{run_id}/.
        limit: Stop after N questions in total (pilot runs); None runs them all.
        notes: What this run changes, recorded in run_metadata.json on creation.
        log_subagent_details: Dump each invocation's report and pre-loaded context under
            runs/{run_id}/debug_v2/. The check verdicts go to runs/{run_id}/
            verifications.jsonl regardless of this flag.
        p: Fraction of a STRATIFIED draw (dataset.sample_dataset), the native form of the
            load_samples monkeypatch a v1 run does from the notebook. None takes every
            question. Match the p and seed of the v1 run you are comparing against and both
            arms answer the same questions.
        seed: Seed of that draw; a same (p, seed) rebuilds the same sample.
    """
    samples = load_samples() if p is None else sample_dataset(p, seed)
    run_dir = _init_run_v2(run_id, notes, log_subagent_details, p, seed)
    done_total = 0

    # doc_ocr.json is the real prerequisite: the DOM is derived from it on first use, and
    # the artifacts are shared with the v1 runs, so both arms read the same frozen DOM.
    for ocr_path in sorted(ARTIFACTS.glob("*/doc_ocr.json")):
        doc_dir = ocr_path.parent
        results = run_dir / f"{doc_dir.name}.jsonl"

        done = set()
        if results.exists():
            done = {json.loads(line)["query_id"]
                    for line in results.read_text(encoding="utf-8").splitlines()
                    if line.strip()}

        pending = [s for s in samples
                   if s["stem"] == doc_dir.name and s["query_id"] not in done]
        if limit is not None:
            pending = pending[:limit - done_total]
        if not pending:
            continue

        # Loaded or built once per document, not once per question. The PDF is resolved
        # here too: its presence is a property of the document, not of the question.
        dom, skeleton = load_or_build_dom(doc_dir)
        pdf_path = pdf_for(doc_dir.name)

        print(f"[{run_id}] [{doc_dir.name}] {len(pending)} questions, {len(done)} déjà faites")

        # Appended and flushed per question: a crash keeps every answer already paid for.
        with results.open("a", encoding="utf-8") as out:
            bar = tqdm(pending, desc=doc_dir.name[:30], unit="q")
            for sample in bar:
                bar.set_postfix_str(sample["query_id"][-8:])
                # An endpoint failure is an ABSENCE of measurement, so it writes no row:
                # the resume filter keys on query_id and replays the question on the next
                # launch. A row would freeze it as done — and here it would render as
                # "Not answerable" and score 1.0 against an unanswerable gold.
                verifications: list = []
                try:
                    row, debug = run_one(dom, skeleton, sample, pdf_path,
                                         log_subagent_details, verifications)
                except Exception as e:
                    if not is_infra_failure(e):
                        raise
                    log_failure(run_dir, sample["query_id"], e)
                    continue
                out.write(json.dumps(row, ensure_ascii=False) + "\n")
                out.flush()
                if verifications:
                    _dump_verifications(run_dir, sample["query_id"], verifications)
                if debug:
                    _dump_debug(run_dir, sample["query_id"], debug)

        done_total += len(pending)
        if limit is not None and done_total >= limit:
            break


def run_and_score(run_id: str, limit: int | None = None, notes: str = "",
                  log_subagent_details: bool = False,
                  p: float | None = None, seed: int = 42) -> Path:
    """The whole chain in one call: run, aggregate, extract, score.

    The three post-hoc stages are run_mm's, extract's and score_mm's own, unchanged: a v2
    run is aggregated, extracted and scored exactly like a v1 run.

    Args:
        run_id: Name of this iteration.
        limit: Stop after N questions in total; None runs them all.
        notes: What this run changes.
        log_subagent_details: Dump the heavy per-invocation artefacts.
        p: Stratified sampling fraction, or None for every question.
        seed: Seed of that draw.

    Returns:
        Path of the scoring output.
    """
    run_mm_v2(run_id, limit=limit, notes=notes,
              log_subagent_details=log_subagent_details, p=p, seed=seed)
    aggregate_run(run_id)
    extract_run(run_id)
    return score_run(run_id)








"""MMLongBench-Doc runner of the v2 agent — run_mm, repointed at agent_dom_v2.

Twin of eval/finragbench/run_v2.py, differing only where run_mm already differs from
run_benchmark: this benchmark's dataset and its own prompt bank, a gold-free row joined by
query_id at scoring time, and a question-to-document match on an explicit `stem` field
rather than a prefix test on the query id.

Nothing in run_mm is modified: its constants, load_or_build_dom, pdf_for and aggregate_run
are imported and reused, and the scoring chain stays extract_run then score_run.

The verdict records of the agent's non-repairing checks (key_figures, page ranges) go to
runs/{run_id}/verifications.jsonl, one line per verdict, next to the per-document files;
aggregate_run lists that name among its derived files, so it never folds into all.jsonl.

SAMPLING. `p` draws the same stratified subsample (on answerability x evidence sources)
that a v1 run takes by monkeypatching dataset.load_samples from the notebook:

    echantillon = sample_dataset(0.3); rm.load_samples = lambda: echantillon

Passing p here does that natively, and records `sample_p` / `sample_seed` in
run_metadata.json — which the monkeypatch could not, leaving a v1 run's question set
recoverable only from its free-text notes. sample_dataset is deterministic for a given
(p, seed), so p=0.3 rebuilds exactly the draw of an mm_strat30 run and the two arms answer
the same questions. p=None takes every question; `limit` then truncates from the head,
document by document in sorted order, in both cases.
"""

def run_and_score(run_id: str, limit: int | None = None, notes: str = "",
                  log_subagent_details: bool = False,

    """The whole chain in one call: run, aggregate, extract, score.

    The three post-hoc stages are run_mm's, extract's and score_mm's own, unchanged: a v2
    run is aggregated, extracted and scored exactly like a v1 run.

    Args:
        run_id: Name of this iteration.
        limit: Stop after N questions in total; None runs them all.
        notes: What this run changes.
        log_subagent_details: Dump the heavy per-invocation artefacts.
        p: Stratified sampling fraction, or None for every question.
        seed: Seed of that draw.

    Returns:
        Path of the scoring output.
    """
    run_mm_v2(run_id, limit=limit, notes=notes,
              log_subagent_details=log_subagent_details, p=p, seed=seed)
    aggregate_run(run_id)
    extract_run(run_id)
    return score_run(run_id)
