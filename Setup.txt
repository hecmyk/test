# --- 1. Setup ---------------------------------------------------------------
import sys; sys.path.insert(0, "eval")     # depuis la racine du repo
from pathlib import Path
from score_results import score_results
import analyze

RUN_ID = "run2_no_nearby"



# --- 2. Scoring -------------------------------------------------------------
# Le cache est keyé sur query_id seul : il ignore le changement de juge. Sans cette
# suppression, un run déjà scoré renverrait les jugements de l'ancien prompt binaire,
# sans un seul appel API et sans rien signaler.
Path(f"eval/runs/{RUN_ID}/judge_cache.json").unlink(missing_ok=True)

score_results(RUN_ID)
df = analyze.load_scored(RUN_ID)
df["outcome"].value_counts()



# --- 3. Sanity check du juge ------------------------------------------------
# À lire AVANT tout le reste : un juge qui ne rend pas de JSON valide est scoré None,
# et l'accuracy est alors calculée sur un sous-ensemble.
failed = df[df["judge_reasoning"].str.startswith(("json_parse_failed", "invalid_score_value"))]
print(f"{len(failed)} / {len(df)} jugements illisibles")
failed[["query_id", "judge_reasoning"]]




# --- 4. Vue d'ensemble ------------------------------------------------------
analyze.overview(df)


# --- 5. Coût : tool calls et tokens ----------------------------------------
from IPython.display import display

display(analyze.cost_by(df, "outcome"))
display(analyze.cost_by(df[df["cited"]], "page_ok"))   # citations seules, sans abstentions



# --- 6. Qualité par catégorie et par type de réponse ------------------------
display(analyze.quality_by(df, "category"))
display(analyze.quality_by(df, "answer_type"))



# --- 7. Tool calls refusés --------------------------------------------------
inv = analyze.invalid_calls(df)
print(f"{len(inv)} refus sur {inv['query_id'].nunique()} queries")
display(inv["form"].value_counts())
display(inv["reason"].value_counts())
inv.head(20)



# --- 8. Figures -------------------------------------------------------------
analyze.plot_cost_by_outcome(df)
analyze.plot_recall_by_category(df, min_n=10)




# --- 9. Ad hoc --------------------------------------------------------------
# Bonne page, mauvaise réponse : problème de lecture, pas de recherche.
misread = df[(df["outcome"] == "wrong") & df["page_ok"]]
print(f"lu au bon endroit mais faux : {len(misread)}")
display(misread[["query_id", "category", "gold_answer", "pred_answer", "judge_reasoning"]].head(15))

# Réponses arrachées au plafond de MAX_STEPS.
display(df[df["submit_draft_source"] == "forced"][["query_id", "outcome", "n_get_section_calls"]])

# Le 2×2 recherche vs lecture, sur les lignes ayant cité une page.
import pandas as pd
pd.crosstab(df[df["cited"]]["judge_score"], df[df["cited"]]["page_ok"])
