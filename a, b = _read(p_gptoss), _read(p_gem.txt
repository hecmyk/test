from pathlib import Path
import json, collections

run = Path("eval/runs/<gptoss>")
seen = collections.defaultdict(list)
for p in sorted(run.glob("*.jsonl")):
    if p.name in {"all.jsonl", "scored.jsonl"}:
        continue
    for i, l in enumerate(p.read_text(encoding="utf-8").splitlines()):
        if l.strip():
            seen[json.loads(l)["query_id"]].append((p.name, i))

for qid, locs in seen.items():
    if len(locs) > 1:
        print(qid, locs)
print("total lignes :", sum(len(v) for v in seen.values()), "| query_id uniques :", len(seen))
