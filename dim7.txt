dim7 and dim8 come from the pointer network cascaded on top of RT-DETR in PP-DocLayoutV2, which predicts the reading order of the detected elements.

The documentation says reading order is encoded as a (col, row) pair, sort by column then row to get the linear order. Empirically on our export this doesn't hold: dim7 is unique per bbox (no repetitions across boxes in the same column), and dim8's role is unclear. I sort by dim7 alone; dim8 is ignored.




# 4) récap global (aucune stat par document)
import statistics
from collections import Counter

READ_TOOLS = ("get_page", "get_crop")


def _base(tool_name: str) -> str:
    """'get_page (text-recovered)' -> 'get_page'."""
    return tool_name.split(" ")[0]


ok = [e for e in entries if "agent" in e]
failed = [e for e in entries if "agent" not in e]

print("=" * 64)
print(f"RÉCAP GLOBAL — {len(ok)}/{len(entries)} docs OK, {len(failed)} en échec")
for e in failed:
    print(f"  ÉCHEC {e['doc_id']}: {e['error']}")

if not ok:
    print("  aucun document traité avec succès, rien à agréger.")
else:
    prep = [e["preprocessing_time_s"] for e in ok]
    agt = [e["agent"]["agent_time_s"] for e in ok]
    tokens, by_tool, by_base, salvaged = Counter(), Counter(), Counter(), Counter()
    read_per_doc, forced, repeats = [], 0, 0

    for e in ok:
        a = e["agent"]
        tokens.update(a["tokens"])
        for name, n in a["tool_calls_by_type"].items():
            by_tool[name] += n
            by_base[_base(name)] += n
            if "text-recovered" in name:
                salvaged[_base(name)] += n
            if "forced" in name:
                forced += n
        reads = [t for t in a["tool_log"] if _base(t["tool"]) in READ_TOOLS]
        read_per_doc.append(len(reads))
        seen = Counter((t["tool"], str(t["args"])) for t in reads)
        repeats += sum(n - 1 for n in seen.values() if n > 1)

    n_pages = sum(e["n_pages"] for e in ok)
    n_verified = sum(e["agent"]["n_pages_verified"] for e in ok)
    n_corr = sum(e["agent"]["n_corrections"] for e in ok)

    print(f"  pages traitées         : {n_pages}  ({n_pages / len(ok):.1f} / doc)")
    print(f"  preprocessing          : {sum(prep):.1f} s  ({statistics.mean(prep):.2f} s / doc)")
    print(f"  agent                  : {sum(agt):.1f} s  ({statistics.mean(agt):.2f} s / doc)")
    print(f"    phase 1 / phase 2    : {sum(e['agent']['phase1_time_s'] for e in ok):.1f} s"
          f" / {sum(e['agent']['phase2_time_s'] for e in ok):.1f} s")
    print(f"  total                  : {sum(e['total_time_s'] for e in ok):.1f} s")
    print(f"  tokens                 : {tokens['total']}"
          f"  (prompt {tokens['prompt']}, completion {tokens['completion']})")
    print(f"  turns                  : {sum(e['agent']['n_turns'] for e in ok)}")

    print("-" * 64)
    print("TOOL CALLS")
    print(f"  docs ayant appelé un outil de lecture : "
          f"{sum(1 for r in read_per_doc if r)}/{len(ok)}")
    print(f"  read calls (get_page + get_crop)      : {sum(read_per_doc)}"
          f"   min {min(read_per_doc)} / méd {statistics.median(read_per_doc):.0f}"
          f" / max {max(read_per_doc)} par doc")
    print(f"    get_page                            : {by_base['get_page']}")
    print(f"    get_crop                            : {by_base['get_crop']}")
    print(f"  appels répétés (même outil + args)    : {repeats}")
    print(f"  submit_draft                          : {by_base['submit_draft']}")
    print(f"  submit_page_corrections               : {by_base['submit_page_corrections']}")
    print(f"  détail brut par type                  : {dict(by_tool)}")

    print("-" * 64)
    print("SANTÉ ENDPOINT / AGENT")
    print(f"  tool calls récupérés depuis du texte  : {dict(salvaged) or '—'}")
    print(f"  submit_draft forcés (max_steps)       : {forced}")
    print(f"  docs hit_max_iterations               : "
          f"{sum(e['agent']['hit_max_iterations'] for e in ok)}/{len(ok)}")

    print("-" * 64)
    print("PHASE 2 (vérification visuelle)")
    print(f"  pages vérifiées                       : {n_verified}")
    rate = f"  ({n_corr / n_verified:.0%} des pages vérifiées)" if n_verified else ""
    print(f"  corrections retenues                  : {n_corr}{rate}")

    filled = Counter()
    for e in ok:
        for f, v in (e.get("result") or {}).items():
            filled[f] += int(v.get("value") not in (None, ""))
    if filled:
        print("-" * 64)
        print(f"CHAMPS RENSEIGNÉS (sur {len(ok)} docs)")
        for f, n in sorted(filled.items(), key=lambda kv: -kv[1]):
            print(f"  {f:<40} {n}/{len(ok)}")
