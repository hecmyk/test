def _clamp_bbox(bbox, width, height, pad: int = 0):
    try:
        x1, y1, x2, y2 = (int(v) for v in bbox)
    except (TypeError, ValueError):
        return None
    if x2 - x1 < _MIN_CROP_PX or y2 - y1 < _MIN_CROP_PX:   # ← AJOUT : sur la bbox BRUTE
        return None

    x1, y1, x2, y2 = x1 - pad, y1 - pad, x2 + pad, y2 + pad   # ← AJOUT
    x1, x2 = (max(0, min(v, width)) for v in (x1, x2))
    y1, y2 = (max(0, min(v, height)) for v in (y1, y2))
    if x2 - x1 < _MIN_CROP_PX or y2 - y1 < _MIN_CROP_PX:   # inchangé (boîte hors page)
        return None
    return x1, y1, x2, y2
